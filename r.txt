rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isOwner() {
      return request.auth != null && request.auth.token.email.lower() == "kalzunni@gmail.com";
    }

    match /products/{id} {
      allow read: if true;
      allow create, delete: if isOwner();
      allow update: if isOwner() || (
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['stock']) &&
        request.resource.data.stock is int &&
        request.resource.data.stock >= 0 &&
        request.resource.data.stock < resource.data.stock
      );
    }

    match /orders/{id} {
      allow create: if request.resource.data.status == 'new'
                    && request.resource.data.createdAt == request.time
                    && request.resource.data.lines is list
                    && request.resource.data.lines.size() > 0;
      allow read, update, delete: if isOwner();
    }

    match /prep/{orderId} {
      allow read: if true;
      allow create, update: if request.resource.data.keys().hasOnly(['done','total','ready','at'])
        && request.resource.data.done is list
        && request.resource.data.done.size() <= 200;
      allow delete: if isOwner();
    }
  }
}
